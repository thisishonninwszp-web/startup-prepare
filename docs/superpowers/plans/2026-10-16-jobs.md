# 求职（/jobs）第 1 块 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增一级模块 `/jobs`「求职」：底稿访谈、岗位（手机分享 / 手动加 / 每天搜）、对照底稿、按岗位改職務経歴書（每句有出处、代码核对）、投递记录和"市场怎么看我"的计数。

**Architecture:** 纯规则（类型、年收差额、地点过滤、去重、四类轮流、没回音、计数、出处核对、招聘页正文抽取）放 `lib/domains/jobs/*-shared.ts` / `page-text.ts`，全部 vitest 测；AI prompt 与解析器放 `lib/ai/jobs.ts`（经桶导出、走 `lib/ai-gateway.ts`）；读写数据库放 `lib/domains/jobs/{profile,base,postings,search,resumes,applications}.ts`；页面在 `app/(app)/jobs/`，打印页在无壳的 `app/(print)/resume-print/`。每天的搜索在第一次打开 `/jobs` 时由 server action 跑（7 点的定时任务已经没有余量：10-13~10-16 的报告都在 7:00:50–7:01:22 才写完）。

**Tech Stack:** Next.js 14 App Router、TypeScript、Supabase（service role）、Gemini（经 `executeAiJson`）、vitest、Tailwind 语义 token。

**Spec:** `docs/superpowers/specs/2026-10-16-jobs-design.md`

**验证命令（每个 task 末尾按需跑）：** `npx tsc --noEmit`、`npm run lint`、`npx vitest run <file>`；最后 `npm test`。**绝不在 dev server 运行时 `npm run build`。**

**与 spec 的两处偏离：**
1. spec 写"搭 7 点的定时任务、不够再打开页面时做"。查库发现定时任务已经吃满 60 秒，所以本计划只做"每天第一次打开 `/jobs` 时搜"。
2. spec 写底稿可以"从材料箱挑一份当素材"。这一版只做"把文字贴进『你的话』"，从材料箱挑等真用到时再加。

```sql
-- ============================================================================
-- 迁移 078：求职（/jobs，经用户明确裁决新增的一级模块，2026-10）
-- job_profiles：条件（四类、地点、现年收、期望下限、在留资格）和个人信息。
--   personal 里是姓名、联系方式、住址，永远不发给 AI，只在打印简历时由代码填进去。
--   last_search_on：今天搜过没有（每天第一次打开 /jobs 时搜）。
-- career_entries：底稿。一段经历一条；raw_text 是他的原话，draft_ja 是 AI 整理的日文；
--   confirmed_at 为空的不算数（对照和改简历只用确认过的）。
-- job_postings：岗位。picked_at 为空的是每天搜到、他还没点"要这个"的候选，14 天后清掉。
--   facts 存抽出来的字段（每条条件都带招聘页原话），match 存跟底稿的对照。
--   同一家公司 + 同一个职位名（dedupe_key）或同一个网址，算同一个岗位。
-- job_resumes：某个岗位的一版简历，每句带出处（底稿的 id）；他改过的段存在 sections 的 user_text 里。
-- job_applications：一次投递，一个岗位最多一条。events 记每次状态变化的日期；
--   "没回音"不存，由代码按 21 天算。
-- 幂等，可重复执行。
-- ============================================================================

create table if not exists job_profiles (
  user_id        uuid primary key references auth.users (id) on delete cascade,
  categories     text[] not null default '{data,consulting,planning,chinese}',
  locations      text[] not null default '{大阪}',
  salary_now     integer,
  salary_floor   integer,
  visa           text not null default 'gijinkoku' check (visa in ('gijinkoku', 'none')),
  personal       jsonb not null default '{}'::jsonb,
  last_search_on date,
  created_at     timestamptz not null default now(),
  updated_at     timestamptz not null default now()
);

alter table job_profiles enable row level security;
drop policy if exists "job_profiles_owner" on job_profiles;
create policy "job_profiles_owner" on job_profiles
  for all using (auth.uid() = user_id);

create table if not exists career_entries (
  id           uuid primary key default gen_random_uuid(),
  user_id      uuid not null references auth.users (id) on delete cascade,
  topic        text not null check (topic in ('education', 'work', 'project', 'language', 'license')),
  org          text not null default '',
  period       text not null default '',
  raw_text     text not null,
  draft_ja     text,
  confirmed_at timestamptz,
  created_at   timestamptz not null default now(),
  updated_at   timestamptz not null default now()
);

create index if not exists idx_career_entries_user on career_entries (user_id, created_at);
alter table career_entries enable row level security;
drop policy if exists "career_entries_owner" on career_entries;
create policy "career_entries_owner" on career_entries
  for all using (auth.uid() = user_id);

create table if not exists job_postings (
  id           uuid primary key default gen_random_uuid(),
  user_id      uuid not null references auth.users (id) on delete cascade,
  company_id   uuid references companies (id) on delete set null,
  company_name text not null,
  title        text not null,
  url          text,
  dedupe_key   text not null,
  source       text not null check (source in ('share', 'search', 'manual', 'email')),
  material_id  uuid references reality_materials (id) on delete set null,
  body         text not null,
  facts        jsonb not null,
  category     text not null check (category in ('data', 'consulting', 'planning', 'chinese', 'none')),
  match        jsonb,
  picked_at    timestamptz,
  found_on     date not null,
  created_at   timestamptz not null default now(),
  updated_at   timestamptz not null default now(),
  constraint job_postings_key_uniq unique (user_id, dedupe_key)
);

create unique index if not exists job_postings_url_uniq on job_postings (user_id, url) where url is not null;
create index if not exists idx_job_postings_user on job_postings (user_id, found_on desc);
alter table job_postings enable row level security;
drop policy if exists "job_postings_owner" on job_postings;
create policy "job_postings_owner" on job_postings
  for all using (auth.uid() = user_id);

create table if not exists job_resumes (
  id         uuid primary key default gen_random_uuid(),
  user_id    uuid not null references auth.users (id) on delete cascade,
  posting_id uuid not null references job_postings (id) on delete cascade,
  version    integer not null,
  sections   jsonb not null,
  probes     jsonb not null default '[]'::jsonb,
  dropped    jsonb not null default '[]'::jsonb,
  created_at timestamptz not null default now(),
  constraint job_resumes_version_uniq unique (posting_id, version)
);

alter table job_resumes enable row level security;
drop policy if exists "job_resumes_owner" on job_resumes;
create policy "job_resumes_owner" on job_resumes
  for all using (auth.uid() = user_id);

create table if not exists job_applications (
  id             uuid primary key default gen_random_uuid(),
  user_id        uuid not null references auth.users (id) on delete cascade,
  posting_id     uuid not null references job_postings (id) on delete cascade,
  resume_id      uuid references job_resumes (id) on delete set null,
  channel        text not null check (channel in ('site', 'board', 'agent', 'email')),
  status         text not null default 'applied'
    check (status in ('applied', 'replied', 'interview', 'rejected', 'withdrawn')),
  events         jsonb not null default '[]'::jsonb,
  offered_salary integer,
  applied_on     date not null,
  created_at     timestamptz not null default now(),
  updated_at     timestamptz not null default now(),
  constraint job_applications_posting_uniq unique (posting_id)
);

alter table job_applications enable row level security;
drop policy if exists "job_applications_owner" on job_applications;
create policy "job_applications_owner" on job_applications
  for all using (auth.uid() = user_id);
```

/**
 * 求职（/jobs）的类型和规则。纯函数，浏览器和服务器都能引。
 * 宪法原则 1：岗位不打分、不算匹配度 —— 这里算的都是"差多少""几家""有几条"，不是分。
 * "市场怎么看我"的数字套用 self 例外的规矩：代码数出来、点开看得到是哪几家、
 * 某类不满 5 家只给次数不给比例、四类之间不合计。
 */

export const JOB_CATEGORIES = {
  data: "数据分析",
  consulting: "咨询",
  planning: "事业企划、经营企划",
  chinese: "需要中文",
} as const;
export type JobCategory = keyof typeof JOB_CATEGORIES;
export type PostingCategory = JobCategory | "none";
export const ALL_CATEGORIES = Object.keys(JOB_CATEGORIES) as JobCategory[];

export function isJobCategory(value: unknown): value is JobCategory {
  return typeof value === "string" && value in JOB_CATEGORIES;
}

export function categoryLabel(category: PostingCategory): string {
  return category === "none" ? "不在你的四类里" : JOB_CATEGORIES[category];
}

export const REMOTE_LEVELS = { full: "全远程", mostly: "基本远程", hybrid: "混合", none: "不远程", unknown: "没写" } as const;
export type RemoteLevel = keyof typeof REMOTE_LEVELS;

export const VISA_FITS = { fits: "对得上", no: "对不上", unclear: "说不清" } as const;
export type VisaFit = keyof typeof VISA_FITS;

export const APPLICATION_STATUSES = {
  applied: "投了",
  replied: "回了",
  interview: "约面试",
  rejected: "不录用",
  withdrawn: "你撤了",
} as const;
export type ApplicationStatus = keyof typeof APPLICATION_STATUSES;
export type EffectiveStatus = ApplicationStatus | "no_reply";
export const EFFECTIVE_LABELS: Record<EffectiveStatus, string> = { ...APPLICATION_STATUSES, no_reply: "没回音" };

export function isApplicationStatus(value: unknown): value is ApplicationStatus {
  return typeof value === "string" && value in APPLICATION_STATUSES;
}

export const CHANNELS = { site: "公司官网", board: "求人网站", agent: "中介", email: "邮件" } as const;
export type Channel = keyof typeof CHANNELS;

export function isChannel(value: unknown): value is Channel {
  return typeof value === "string" && value in CHANNELS;
}

export const CAREER_TOPICS = {
  education: "学历",
  work: "工作",
  project: "自己发起的项目",
  language: "语言",
  license: "资格",
} as const;

/** 年收（万日元）跟现在比差多少。只说差额，不说值不值。 */
export function salaryDiff(now: number | null, min: number | null, max: number | null): string | null {
  if (now === null) return null;
  const known = [min, max].filter((n): n is number => n !== null);
  if (known.length === 0) return null;
  const lo = Math.min(...known);
  const hi = Math.max(...known);
  const span = (a: number, b: number) => (a === b ? `${a}` : `${a}–${b}`);
  if (lo === now && hi === now) return "跟现在一样";
  if (lo >= now) return `比现在多 ${span(lo - now, hi - now)} 万`;
  if (hi <= now) return `比现在少 ${span(now - hi, now - lo)} 万`;
  return `比现在少 ${now - lo} 万到多 ${hi - now} 万`;
}

/** 年收上限都够不着期望下限（软条件：只标出来，不丢）。 */
export function belowFloor(floor: number | null, min: number | null, max: number | null): boolean {
  if (floor === null) return false;
  const top = max ?? min;
  return top !== null && top < floor;
}

export type LocationVerdict = "keep" | "drop" | "unclear";

/** 全远程、基本远程不管在哪都留；勤务地含你写的地方的留；抽不清的保留并标"地点说不清"。 */
export function locationVerdict(location: string, remote: RemoteLevel, places: string[]): LocationVerdict {
  if (remote === "full" || remote === "mostly") return "keep";
  const loc = location.trim();
  if (loc && places.some((p) => p.trim() && loc.includes(p.trim()))) return "keep";
  if (remote === "unknown" || !loc) return "unclear";
  return "drop";
}

const COMPANY_NOISE = /株式会社|有限会社|合同会社|\(株\)/g;
const PUNCT = /[\s【】[\]()「」『』・,、.。:：/|~〜!?-]/g;

function squashName(s: string): string {
  return s.normalize("NFKC").toLowerCase().replace(COMPANY_NOISE, "").replace(PUNCT, "");
}

/** 同一家公司 + 同一个职位名，算同一个岗位。 */
export function postingKey(company: string, title: string): string {

/** 去掉 #、utm_ 之类的跟踪参数和路径结尾的 /，同一页面的不同写法算一个。 */
export function normalizeUrl(url: string): string {
  try {
    const u = new URL(url.trim());
    u.hash = "";
    for (const key of Array.from(u.searchParams.keys())) {
      if (/^(utm_|fbclid|gclid)/i.test(key)) u.searchParams.delete(key);
    }
    if (u.pathname.length > 1 && u.pathname.endsWith("/")) u.pathname = u.pathname.slice(0, -1);
    return u.toString();
  } catch {
    return url.trim();
  }
}

/** 四类轮流取：每类最多 perCat 条，一共最多 cap 条。不在你选的几类里的不取。 */
export function pickCandidates<T extends { category: PostingCategory }>(
  items: T[],
  categories: JobCategory[],
  cap = DAILY_CANDIDATE_CAP,
  perCat = PER_CATEGORY_CAP
): T[] {
  const queues = categories.map((c) => items.filter((i) => i.category === c));
  const picked: T[] = [];
  for (let round = 0; round < perCat; round++) {
    for (const queue of queues) {
      if (picked.length >= cap) return picked;
      const next = queue[round];
      if (next) picked.push(next);
    }
  }
  return picked;
}

/** "没回音"不由人点：状态还是"投了"，而且最后一次变化已经过了 21 天。 */
export function effectiveStatus(
  app: { status: ApplicationStatus; events: ApplicationEvent[]; applied_on: string },
  today: string
): EffectiveStatus {
  if (app.status !== "applied") return app.status;
  const last = app.events.length > 0 ? app.events[app.events.length - 1].on : app.applied_on;
  return daysBetween(last, today) >= NO_REPLY_DAYS ? "no_reply" : "applied";
}

export type MarketRow = {
  category: JobCategory;
  applied: number;
  replied: number;
  interview: number;
  rejected: number;
  noReply: number;
  /** 投满 RATIO_MIN_APPLICATIONS 家才有（百分比整数）；分母是这一类投了几家。 */
  replyRate: number | null;
  salaries: number[];
  salaryRange: [number, number] | null;
  apps: (MarketApp & { effective: EffectiveStatus })[];
};

function everReached(app: MarketApp, statuses: ApplicationStatus[]): boolean {
  return statuses.includes(app.status) || app.events.some((e) => statuses.includes(e.status));
}

/** 四类各一行，只数次数；投满 5 家才给"回了"的比例。四类之间不合计。"回了"包括后来约了面试的。 */
export function marketRows(apps: MarketApp[], categories: JobCategory[], today: string): MarketRow[] {
  return categories.map((category) => {
    const mine = apps.filter((a) => a.category === category).map((a) => ({ ...a, effective: effectiveStatus(a, today) }));
    const replied = mine.filter((a) => everReached(a, ["replied", "interview"])).length;
    const salaries = mine.map((a) => a.offered_salary).filter((s): s is number => typeof s === "number");
    return {
      category,
      applied: mine.length,
      replied,
      interview: mine.filter((a) => everReached(a, ["interview"])).length,
      rejected: mine.filter((a) => a.status === "rejected").length,
      noReply: mine.filter((a) => a.effective === "no_reply").length,
      replyRate: mine.length >= RATIO_MIN_APPLICATIONS ? Math.round((replied / mine.length) * 100) : null,
      salaries,
      salaryRange: salaries.length >= 2 ? [Math.min(...salaries), Math.max(...salaries)] : null,
      apps: mine,
    };
  });
}

/** 必须条件和加分条件各有几条对得上、几条缺。 */
export function matchCounts(
  conditions: PostingCondition[],
  match: PostingMatch
): { required: { have: number; missing: number }; bonus: { have: number; missing: number } } {
  const out = { required: { have: 0, missing: 0 }, bonus: { have: 0, missing: 0 } };
  conditions.forEach((c, i) => {
    const bucket = c.required ? out.required : out.bonus;
    if ((match.entry_ids[i] ?? []).length > 0) bucket.have += 1;
    else bucket.missing += 1;
  });
  return out;
}

function salaryField(value: unknown): number | null | "bad" {

    categories: Array.from(new Set(categories)),
    locations,
    salaryNow,
    salaryFloor,
    visa: form.get("visa") === "none" ? "none" : "gijinkoku",
    personal,
  };
}

import type { CareerEntry, DroppedSentence, SourcedSentence } from "./shared";

/**
 * 出处核对：简历和底稿草稿里每一句写到的数字、英文词（工具名、公司名），都必须在它标的出处里找得到。
 * 找不到的句子删掉，写明删的理由 —— AI 只能重新组织他说过的话，不能替他编经历。
 */

function nfkc(s: string): string {
  return s.normalize("NFKC");
}

/** 句子里的数字："04" 和 "4" 算同一个，"1,200" 算 1200，全角数字也认。 */
export function numbersIn(text: string): string[] {
  const out = new Set<string>();
  for (const m of nfkc(text).matchAll(/\d[\d,]*(?:\.\d+)?/g)) {
    const n = Number(m[0].replace(/,/g, ""));
    if (Number.isFinite(n)) out.add(String(n));
  }
  return Array.from(out);
}

/** 英文写的词（GA4、Tableau、Python、Focus……），小写，至少两个字符。 */
export function latinTermsIn(text: string): string[] {
  const out = new Set<string>();
  for (const m of nfkc(text).matchAll(/[A-Za-z][A-Za-z0-9+#.-]*/g)) {
    const term = m[0].replace(/[.-]+$/, "").toLowerCase();
    if (term.length >= 2) out.add(term);
  }
  return Array.from(out);
}

/** 一句话里的数字和英文词是不是都在出处里。返回第一个对不上的理由；都对得上返回 null。 */
export function unsupported(text: string, source: string): string | null {
  const src = nfkc(source).toLowerCase();
  const srcNumbers = new Set(numbersIn(source));
  for (const n of numbersIn(text)) if (!srcNumbers.has(n)) return `数字「${n}」在底稿里找不到`;
  for (const t of latinTermsIn(text)) if (!src.includes(t)) return `「${t}」在底稿里找不到`;
  return null;
}

/** 一条底稿拿来核对的全部文字：在哪、期间、原话、日文稿。 */
export function entrySourceText(entry: Pick<CareerEntry, "org" | "period" | "raw_text" | "draft_ja">): string {
  return [entry.org, entry.period, entry.raw_text, entry.draft_ja ?? ""].join("\n");
}

 * entries：确认过的底稿，编号 → 这条的全部文字。
 * extra：额外允许用的词（志望動機可以提到招聘页上的词）；只用来核对词，不能代替出处。
 */

  raw: unknown,
  entries: Map<string, string>,
  extra = ""
): { kept: SourcedSentence[]; dropped: DroppedSentence[] } {
  const kept: SourcedSentence[] = [];
  const dropped: DroppedSentence[] = [];
  for (const item of Array.isArray(raw) ? raw : []) {
    if (!item || typeof item !== "object") continue;
    const v = item as Record<string, unknown>;
    const text = typeof v.text === "string" ? v.text.trim() : "";
    if (!text) continue;
    const ids = (Array.isArray(v.entry_ids) ? v.entry_ids : []).filter((id): id is string => typeof id === "string");
    if (ids.length === 0) {
      dropped.push({ text, reason: "没标出处" });
      continue;
    }
    const known = Array.from(new Set(ids.filter((id) => entries.has(id))));
    if (known.length === 0) {
      dropped.push({ text, reason: "标的出处不在确认过的底稿里" });
      continue;
    }
    const reason = unsupported(text, [...known.map((id) => entries.get(id) ?? ""), extra].join("\n"));
    if (reason) {
      dropped.push({ text, reason });
      continue;
    }
    kept.push({ text, entry_ids: known });
  }
  return { kept, dropped };
}

/** 底稿的日文稿：按句切开，数字和英文词在他的原话里找不到的句子删掉。 */
export function checkDraft(draft: string, source: string): { text: string; dropped: DroppedSentence[] } {
  const sentences = draft
    .split(/(?<=[。！？!?])|\n+/)
    .map((s) => s.trim())
    .filter(Boolean);
  const kept: string[] = [];
  const dropped: DroppedSentence[] = [];
  for (const sentence of sentences) {
    const reason = unsupported(sentence, source);
    if (reason) dropped.push({ text: sentence, reason: reason.replace("底稿", "你的原话") });
    else kept.push(sentence);
  }
  return { text: kept.join(""), dropped };
}

/** 招聘页原话是不是真在正文里（去掉空白、统一全半角后逐字比）。4 个字以下的不算。 */
export function quoteIn(quote: string, body: string): boolean {
  const squash = (s: string) => nfkc(s).replace(/\s+/g, "");
  const q = squash(quote);
  return q.length >= 4 && squash(body).includes(q);
}

/**
 * 招聘页的正文。读文章用的 extractSearchText 只留 20 字以上的段落，
 * 招聘页的关键信息偏偏在短格子里（「勤務地」「大阪府大阪市」「年収」「450万円〜」），所以单独写一个：
 * 先取页面里的 JobPosting 结构化数据（很多招聘页为了上 Google 求人检索都会放），
 * 再按顺序取标题、段落、列表、表格和定义列表，表头 / 定义词和后面的值并成一行。
 */

const ENTITIES: Record<string, string> = { amp: "&", lt: "<", gt: ">", quot: '"', apos: "'", nbsp: " " };

export function decodeEntities(s: string): string {
  return s.replace(/&(#x[0-9a-f]+|#\d+|[a-z]+);/gi, (all, code: string) => {
    if (code[0] === "#") {
      const n = code[1].toLowerCase() === "x" ? parseInt(code.slice(2), 16) : parseInt(code.slice(1), 10);
      return Number.isFinite(n) ? String.fromCodePoint(n) : all;
    }
    return ENTITIES[code.toLowerCase()] ?? all;
  });
}

function textOf(html: string): string {
  return decodeEntities(html.replace(/<br\s*\/?>/gi, " ").replace(/<[^>]+>/g, " "))
    .replace(/\s+/g, " ")
    .trim();
}

function isJobPosting(node: Record<string, unknown>): boolean {
  const type = node["@type"];
  return type === "JobPosting" || (Array.isArray(type) && type.includes("JobPosting"));
}

/** 页面里 @type 是 JobPosting 的 JSON-LD，整理成几行字。没有就返回空数组。 */
export function jobPostingJsonLd(html: string): string[] {
  const lines: string[] = [];
  for (const m of html.matchAll(/<script[^>]+application\/ld\+json[^>]*>([\s\S]*?)<\/script>/gi)) {
    let data: unknown;
    try {
      data = JSON.parse(m[1].trim());
    } catch {
      continue;
    }
    const roots = Array.isArray(data) ? data : [data];
    const nodes = roots.flatMap((d) => {
      const graph = (d as { "@graph"?: unknown } | null)?.["@graph"];
      return Array.isArray(graph) ? graph : [d];
    });
    for (const raw of nodes) {
      if (!raw || typeof raw !== "object" || !isJobPosting(raw as Record<string, unknown>)) continue;
      const n = raw as Record<string, unknown>;
      const push = (label: string, value: unknown) => {
        if (typeof value !== "string" && typeof value !== "number") return;
        const text = textOf(String(value));
        if (text) lines.push(`${label}：${text}`);
      };
      const org = n.hiringOrganization as Record<string, unknown> | undefined;
      const place = (Array.isArray(n.jobLocation) ? n.jobLocation[0] : n.jobLocation) as Record<string, unknown> | undefined;
      const addr = place?.address as Record<string, unknown> | undefined;
      push("職種", n.title);
      push("会社", org?.name);
      push(
        "勤務地",
        [addr?.addressRegion, addr?.addressLocality, addr?.streetAddress].filter((x): x is string => typeof x === "string").join("")
      );
      push("勤務形態", n.jobLocationType);
      push("雇用形態", Array.isArray(n.employmentType) ? n.employmentType.join("、") : n.employmentType);
      const salary = n.baseSalary as { currency?: unknown; value?: Record<string, unknown> } | undefined;
      if (salary?.value) {
        const v = salary.value;
        const range = [v.minValue, v.maxValue, v.value].filter((x): x is number => typeof x === "number").join("〜");
        if (range) {
          const unit = typeof v.unitText === "string" ? v.unitText : "";
          push("給与", [range, typeof salary.currency === "string" ? salary.currency : "", unit].filter(Boolean).join(" "));
        }
      }
      push("仕事内容", n.description);
    }
  }
  return lines;
}

const LABEL_TAGS = new Set(["th", "dt"]);
const VALUE_TAGS = new Set(["td", "dd"]);

/** 招聘页的正文（结构化数据在前，页面上的字在后），最多 maxLength 字。 */
export function extractJobText(html: string, maxLength = 15_000): string {
  const lines = jobPostingJsonLd(html);
  const cleaned = html
    .replace(/<script[\s\S]*?<\/script>/gi, " ")
    .replace(/<style[\s\S]*?<\/style>/gi, " ")
    .replace(/<noscript[\s\S]*?<\/noscript>/gi, " ")
    .replace(/<(nav|footer|aside)(?:\s[^>]*)?>[\s\S]*?<\/\1>/gi, " ");
  const seen = new Set(lines);
  let total = lines.join("\n").length;
  let pendingLabel: string | null = null;
  for (const m of cleaned.matchAll(/<(h1|h2|h3|h4|p|li|dt|dd|th|td)(?:\s[^>]*)?>([\s\S]*?)<\/\1>/gi)) {
    const tag = m[1].toLowerCase();
    const text = textOf(m[2]);
    if (!text) continue;
    if (LABEL_TAGS.has(tag)) {
      pendingLabel = text;
      continue;
    }
    const line = VALUE_TAGS.has(tag) && pendingLabel ? `${pendingLabel}：${text}` : text;
    pendingLabel = null;
    if (seen.has(line)) continue;
    seen.add(line);
    lines.push(line);
    total += line.length + 1;
    if (total >= maxLength) break;
  }
  return lines.join("\n").slice(0, maxLength);
}

  - quote：页面上的原话，逐字照抄（10 到 100 字）
  - required：必须条件 true，加分条件 false
- terms：别的条件，只抽这几样：远程或出社的天数、加班（残業）、休日休假、副业可否、试用期。每条 label（中文两三个字）和 quote（原话逐字照抄）。没写的不列。
- category：这份工作主要是哪一类：data（数据分析）、consulting（咨询）、planning（事业企划、经营企划）、chinese（中文写在必须条件里）、none（都不是）。中文是必须条件就选 chinese，否则选工作内容最接近的一类。
- visa：工作内容跟技人国对不对得上：
  - fit：fits（企划、分析、市场、咨询、翻译口译、工程师这类要专业知识的工作）/ no（主要是接客、店头销售、工厂、仓库、清扫、驾驶、护理这类现场作业）/ unclear（看不出来）
  - quote：業務内容的原话，逐字照抄（10 到 150 字）
  - reason：一句中文，说为什么这样判。只说工作内容，不替入管下结论。

原话一律逐字照抄，不翻译、不改字、不补字；找不到原话的条目不要写。

只输出 JSON：{"is_posting":true,"company":"…","title":"…","location":"…","remote":"unknown","salary_min":null,"salary_max":null,"salary_quote":"","conditions":[{"text":"…","quote":"…","required":true}],"terms":[{"label":"…","quote":"…"}],"category":"data","visa":{"fit":"fits","quote":"…","reason":"…"}}`;

/** 原话对不上的条件、别的条件丢掉；年收的数要在年收原话里（写成円的也认）；签证没有原话撑着算说不清。 */
export function parsePostingFacts(value: unknown, body: string): ExtractedPosting {
  const v = (value ?? {}) as Record<string, unknown>;
  if (v.is_posting !== true) return { is_posting: false };
  const company = str(v.company, 80);
  const title = str(v.title, 120);
  if (!company || !title) throw new Error("是招聘页但公司名或职位名没写");

  const remote: RemoteLevel = typeof v.remote === "string" && v.remote in REMOTE_LEVELS ? (v.remote as RemoteLevel) : "unknown";

  const salaryQuote = str(v.salary_quote, 200);
  let salaryMin = intOrNull(v.salary_min);
  let salaryMax = intOrNull(v.salary_max);
  const quoted = new Set(numbersIn(salaryQuote));
  const inQuote = (n: number | null) => n === null || quoted.has(String(n)) || quoted.has(String(n * 10_000));
  if (!quoteIn(salaryQuote, body) || !inQuote(salaryMin) || !inQuote(salaryMax)) {
    salaryMin = null;
    salaryMax = null;
  }

  const conditions: PostingCondition[] = (Array.isArray(v.conditions) ? v.conditions : [])
    .flatMap((c) => {
      const x = (c ?? {}) as Record<string, unknown>;
      const text = str(x.text, 80);
      const quote = str(x.quote, 200);
      return text && quoteIn(quote, body) ? [{ text, quote, required: x.required === true }] : [];
    })
    .slice(0, 12);

  const terms: PostingTerm[] = (Array.isArray(v.terms) ? v.terms : [])
    .flatMap((t) => {
      const x = (t ?? {}) as Record<string, unknown>;
      const label = str(x.label, 10);
      const quote = str(x.quote, 200);
      return label && quoteIn(quote, body) ? [{ label, quote }] : [];
    })
    .slice(0, 8);

  const category: PostingCategory =
    v.category === "none" || (typeof v.category === "string" && v.category in JOB_CATEGORIES) ? (v.category as PostingCategory) : "none";

  const visaRaw = (v.visa ?? {}) as Record<string, unknown>;
  const visaQuote = str(visaRaw.quote, 300);
  const quoteOk = quoteIn(visaQuote, body);
  const fit: VisaFit = quoteOk && typeof visaRaw.fit === "string" && visaRaw.fit in VISA_FITS ? (visaRaw.fit as VisaFit) : "unclear";

  return {
    is_posting: true,
    company,
    title,
    category,
    facts: {
      location: str(v.location, 80),
      remote,
      salary_min: salaryMin,
      salary_max: salaryMax,
      salary_quote: salaryMin === null && salaryMax === null ? "" : salaryQuote,
      conditions,
      terms,
      visa: { fit, quote: quoteOk ? visaQuote : "", reason: str(visaRaw.reason, 120) },
    },
  };
}

/** 给 AI 看的底稿用短编号（E1、E2……），比 uuid 好抄；解析完再换回真 id。 */
function aliasEntries(entries: CareerEntry[]): { toId: Map<string, string>; sources: Map<string, string>; block: string } {
  const toId = new Map<string, string>();
  const sources = new Map<string, string>();
  const lines = entries.map((e, i) => {
    const alias = `E${i + 1}`;
    toId.set(alias, e.id);
    sources.set(alias, entrySourceText(e));
    return [
      `[${alias}] ${CAREER_TOPICS[e.topic]}｜${e.org || "（没写在哪）"}｜${e.period || "（没写期间）"}`,
      `原话：${e.raw_text}`,
      e.draft_ja ? `日文稿：${e.draft_ja}` : "",
    ]
      .filter(Boolean)
      .join("\n");
  });
  return { toId, sources, block: lines.join("\n\n") };
}

function conditionsBlock(conditions: PostingCondition[]): string {
  return conditions.map((c, i) => `[${i + 1}]（${c.required ? "必须" : "加分"}）${c.text} —— 原话：「${c.quote}」`).join("\n");
}

const MATCH_PROMPT = `他在找工作。下面是一个岗位的应募条件（带编号），和他确认过的经历底稿（带编号 E1、E2……）。

再写 blocker：这家最可能在哪一条上把他筛掉，一两句中文，直说，不安慰、不鼓励。index 写是第几条。

只输出 JSON：{"matches":[{"index":1,"entry_ids":["E1"]}],"blocker":{"index":2,"text":"…"}}`;

/** 返回的 entry_ids 是编号（E1……），由调用方换回 id。不认识的编号丢掉。 */
export function parseMatch(value: unknown, conditionCount: number, aliases: Set<string>): PostingMatch {
  const v = (value ?? {}) as Record<string, unknown>;
  const ids: string[][] = Array.from({ length: conditionCount }, () => []);
  for (const m of Array.isArray(v.matches) ? v.matches : []) {
    const x = (m ?? {}) as Record<string, unknown>;
    const index = Number(x.index);
    if (!Number.isInteger(index) || index < 1 || index > conditionCount) continue;
    const found = (Array.isArray(x.entry_ids) ? x.entry_ids : []).filter((id): id is string => typeof id === "string" && aliases.has(id));
    ids[index - 1] = Array.from(new Set(found));
  }
  const b = (v.blocker ?? {}) as Record<string, unknown>;
  const text = str(b.text, 200);
  if (!text) throw new Error("blocker 没写");
  assertNoPraise(text);
  const index = Number(b.index);
  return { entry_ids: ids, blocker: { index: Number.isInteger(index) && index >= 1 && index <= conditionCount ? index : 0, text } };
}

/** 每条条件对得上底稿哪几条，以及最可能卡他的一条。 */
export async function matchPosting(userId: string, conditions: PostingCondition[], entries: CareerEntry[]): Promise<PostingMatch> {
  if (conditions.length === 0) {
    return { entry_ids: [], blocker: { index: 0, text: "招聘页上没抽到应募条件，对照不了。打开原页面自己看一眼。" } };
  }
  const { toId, block } = aliasEntries(entries);
  const result = await executeAiJson(
    { userId, operation: "job_match", module: "jobs", outputMode: "json", timeoutMs: 30_000 },
    (attempt) => ({
      model: MODEL,
      contents:
        `应募条件：\n${conditionsBlock(conditions)}\n\n底稿：\n${block}` +
        (attempt === 1 ? "\n\n上一次输出没通过校验：blocker.text 必须写，不许夸他。" : ""),
      config: {
        systemInstruction: MATCH_PROMPT,
        responseMimeType: "application/json",
        thinkingConfig: { thinkingBudget: 0 },
        maxOutputTokens: 1500,
      },
    }),
    (value) => parseMatch(value, conditions.length, new Set(toId.keys()))
  );
  return { ...result, entry_ids: result.entry_ids.map((list) => list.map((alias) => toId.get(alias) ?? alias)) };
}

const TAILOR_PROMPT = `他要投下面这个岗位。给你岗位信息（公司、职位、条件、招聘页节选）和他确认过的经历底稿（编号 E1、E2……）。
用日文写一份職務経歴書和志望動機，分五段，每段若干句，每句都标出处：
- summary（職務要約）：2 到 3 句
- history（職務経歴）：按底稿里的工作和项目写，每条经历 1 到 3 句
- skills（活かせる経験・知識・スキル）：3 到 6 句，每句一项
- pr（自己PR）：2 到 4 句
- motivation（志望動機）：2 到 4 句，把他的经历和这家的工作连起来

规矩：
- 每句的 entry_ids 写这句话出自底稿哪几条（编号）。没有底稿撑着的话不要写。
- 数字、工具名、公司名只能用底稿里写着的，一个字不改；底稿里没有的不许补，"約""以上""大幅"这类他没说的修饰也不许加。
- 朝这个岗位的条件去挑底稿、排顺序，但不许把他没做过的事写成做过。
- 志望動機可以提到这家招聘页上写的工作内容。

再写 probes：面试官看到这份简历最可能追问的 3 句（照抄你写的原句），每句配一个他可能会问的问题（中文）。挑最容易被问穿的，不要挑好回答的。

只输出 JSON：{"summary":[{"text":"…","entry_ids":["E1"]}],"history":[],"skills":[],"pr":[],"motivation":[],"probes":[{"sentence":"…","question":"…"}]}`;

/** 每段各自核对出处；志望動機可以用招聘页上的词。entry_ids 是编号，由调用方换回 id。 */
export function parseTailored(value: unknown, sources: Map<string, string>, postingBody: string): TailoredResume {
  const v = (value ?? {}) as Record<string, unknown>;
  const sections = {} as Record<ResumeSectionKey, ResumeSection>;
  const dropped: TailoredResume["dropped"] = [];
  for (const key of RESUME_SECTION_KEYS) {
    const checked = checkSentences(v[key], sources, key === "motivation" ? postingBody : "");
    sections[key] = { sentences: checked.kept };
    dropped.push(...checked.dropped.map((d) => ({ ...d, section: key })));
  }
  if (sections.summary.sentences.length === 0 && sections.history.sentences.length === 0) {
    throw new Error("職務要約和職務経歴一句都没留下来");
  }
  const keptTexts = new Set(RESUME_SECTION_KEYS.flatMap((k) => sections[k].sentences.map((s) => s.text)));
  const probes = (Array.isArray(v.probes) ? v.probes : [])
    .flatMap((p) => {
      const x = (p ?? {}) as Record<string, unknown>;
      const sentence = str(x.sentence, 300);
      const question = str(x.question, 150);
      return keptTexts.has(sentence) && question ? [{ sentence, question }] : [];
    })
    .slice(0, 3);
  return { sections, probes, dropped };
}

export async function tailorResume(
  userId: string,
  posting: { company: string; title: string; conditions: PostingCondition[]; body: string },
  entries: CareerEntry[]
): Promise<TailoredResume> {
  const { toId, sources, block } = aliasEntries(entries);
  const contents = [
    `岗位：${posting.company}｜${posting.title}`,
    `条件：\n${conditionsBlock(posting.conditions) || "（没抽到）"}`,
    `招聘页节选：\n${posting.body.slice(0, 4000)}`,
    `底稿：\n${block}`,
  ].join("\n\n");
  const result = await executeAiJson(
    { userId, operation: "job_tailor", module: "jobs", outputMode: "json", timeoutMs: 50_000 },
    (attempt) => ({
      model: MODEL,
      contents:
        contents +
        (attempt === 1 ? "\n\n上一次输出没通过校验：每句都要标底稿编号，数字和英文词只能用底稿里写着的。" : ""),
      config: {
        systemInstruction: TAILOR_PROMPT,
        responseMimeType: "application/json",
        thinkingConfig: { thinkingBudget: 0 },
        maxOutputTokens: 4000,
      },
    }),
    (value) => parseTailored(value, sources, posting.body)
  );
  const sections = Object.fromEntries(
    RESUME_SECTION_KEYS.map((key) => [
      key,
      { sentences: result.sections[key].sentences.map((s) => ({ ...s, entry_ids: s.entry_ids.map((alias) => toId.get(alias) ?? alias) })) },
    ])
  ) as Record<ResumeSectionKey, ResumeSection>;
  return { ...result, sections };
}

    org,
    period,
    raw_text: raw,
    draft_ja: draft.draft_ja,
    confirmed_at: null,
    updated_at: new Date().toISOString(),
  };
  if (input.id) {
    const { error } = await supabaseAdmin.from("career_entries").update(row).eq("id", input.id).eq("user_id", userId);

async function findExisting(userId: string, url: string | null, key: string): Promise<string | null> {

  userId: string,
  extracted: Extract<ExtractedPosting, { is_posting: true }>,
  input: { body: string; url: string | null; source: PostingSource; materialId?: string | null; picked: boolean; foundOn: string }
): Promise<SavedPosting> {
  const key = postingKey(extracted.company, extracted.title);
  const existing = await findExisting(userId, input.url, key);
  if (existing) return { id: existing, existed: true };
  const companyId = input.picked ? await ensureProspectCompany(userId, extracted.company) : null;
  const { data, error } = await supabaseAdmin
    .from("job_postings")
    .insert({
      user_id: userId,
      company_id: companyId,
      company_name: extracted.company,
      title: extracted.title,

      source: input.source,

      body: input.body,

      picked_at: input.picked ? new Date().toISOString() : null,
      found_on: input.foundOn,

/** 一页正文 → 岗位（分享、手动加、以后的邮件都走这里），马上挂公司、跟底稿对照。 */
export async function createPosting(
  userId: string,
  input: { body: string; url: string | null; source: PostingSource; materialId?: string | null }
): Promise<CreatePostingResult> {
  const extracted = await extractPostingFacts(userId, input.body);
  if (!extracted.is_posting) return { ok: false, reason: "这一页看起来不是一个具体职位的招聘页（列表页、文章这类不算）。" };

  if (saved.existed) await pickPosting(userId, saved.id);
  else await refreshMatchQuietly(userId, saved.id);
  return { ok: true, ...saved };
}

/** 跟确认过的底稿对照。底稿一条都没确认时返回 "no-entries"，不调 AI。 */
export async function refreshMatch(userId: string, id: string): Promise<PostingMatch | "no-entries"> {

  if (!posting) throw new Error("找不到这个岗位");

  if (entries.length === 0) return "no-entries";
  const match = await matchPosting(userId, posting.facts.conditions, entries);
  const { error } = await supabaseAdmin

    .update({ match, updated_at: new Date().toISOString() })
    .eq("id", id)
    .eq("user_id", userId);

  return match;
}

}

import { webSearch } from "@/lib/web-search";
import { extractPostingFacts, type ExtractedPosting } from "@/lib/ai";
import { getJobProfile, markSearched, MISSING_MIGRATION } from "./profile";
import { cleanOldCandidates, knownUrls, readJobPage, savePosting } from "./postings";
import { JOB_CATEGORIES, locationVerdict, normalizeUrl, pickCandidates, searchQuery, tokyoDate } from "./shared";

/**
 * 每天搜岗位：每类搜一次（最多 4 次），每类读前 3 页，抽岗位、按地点过滤、四类轮流留最多 8 条候选。
 * 在打开 /jobs 的那个 server action 里跑（60 秒）：7 点的定时任务已经没有余量。
 * AI 抽岗位一次最多并发 4 个，免得撞 Gemini 的限流。
 */

const PAGES_PER_CATEGORY = 3;
const CONCURRENCY = 4;

async function mapLimit<T, R>(items: T[], limit: number, fn: (item: T) => Promise<R>): Promise<R[]> {
  const out: R[] = [];
  for (let i = 0; i < items.length; i += limit) {
    out.push(...(await Promise.all(items.slice(i, i + limit).map(fn))));
  }
  return out;
}

type Analyzed = { url: string; body: string; extracted: Extract<ExtractedPosting, { is_posting: true }>; category: Extract<ExtractedPosting, { is_posting: true }>["category"] };

export async function runDailyJobSearch(userId: string, today = tokyoDate()): Promise<{ read: number; kept: number; skipped: string[] }> {
  const profile = await getJobProfile(userId);
  if (profile === "missing-schema") throw new Error(MISSING_MIGRATION);
  if (!profile) throw new Error("先在「条件」里把要找的几类和地点填好。");
  // 先记上今天搜过：两个标签页同时打开不会搜两遍；没搜成的话页面上有"再搜一次"
  await markSearched(userId, today);
  await cleanOldCandidates(userId, today);

  const skipped: string[] = [];
  const perCategory = await Promise.all(
    profile.categories.map(async (category) => {
      try {
        const results = await webSearch(searchQuery(category, profile.locations), 5);
        return results.slice(0, PAGES_PER_CATEGORY).map((r) => normalizeUrl(r.url));
      } catch {
        skipped.push(`${JOB_CATEGORIES[category]}：没搜成`);
        return [];
      }
    })
  );
  const urls = Array.from(new Set(perCategory.flat()));
  const known = await knownUrls(userId, urls);
  const fresh = urls.filter((u) => !known.has(u));

  const analyzed = (
    await mapLimit(fresh, CONCURRENCY, async (url): Promise<Analyzed | null> => {
      const page = await readJobPage(url).catch(() => null);
      if (!page || !page.ok) return null;
      try {
        const extracted = await extractPostingFacts(userId, page.body);
        return extracted.is_posting ? { url: page.url, body: page.body, extracted, category: extracted.category } : null;
      } catch {
        return null;
      }
    })
  )
    .filter((x): x is Analyzed => x !== null)
    .filter((x) => locationVerdict(x.extracted.facts.location, x.extracted.facts.remote, profile.locations) !== "drop");

  let kept = 0;
  for (const item of pickCandidates(analyzed, profile.categories)) {
    const saved = await savePosting(userId, item.extracted, { body: item.body, url: item.url, source: "search", picked: false, foundOn: today });
    if (!saved.existed) kept += 1;
  }
  return { read: analyzed.length, kept, skipped };
}

  type PostingCategory,

function toEvents(value: unknown): ApplicationEvent[] {
  return (Array.isArray(value) ? value : []).flatMap((e) => {
    const x = (e ?? {}) as Record<string, unknown>;
    return isApplicationStatus(x.status) && typeof x.on === "string" ? [{ status: x.status, on: x.on }] : [];
  });
}

export async function getApplicationForPosting(userId: string, postingId: string): Promise<JobApplication | null> {
  const { data, error } = await supabaseAdmin
    .from("job_applications")
    .select("id, posting_id, resume_id, channel, status, events, offered_salary, applied_on")
    .eq("posting_id", postingId)
    .eq("user_id", userId)
    .maybeSingle();
  if (error) {
    if (isMissingJobTables(error.message)) return null;
    throw new Error(error.message);
  }
  if (!data) return null;
  return {
    id: data.id as string,
    posting_id: data.posting_id as string,
    resume_id: (data.resume_id as string | null) ?? null,
    channel: data.channel as Channel,
    status: data.status as ApplicationStatus,
    events: toEvents(data.events),
    offered_salary: (data.offered_salary as number | null) ?? null,
    applied_on: data.applied_on as string,
  };
}

/** 记一次投递（一个岗位只记一次）。还没点"要这个"的候选，顺手变成要了的。 */
export async function recordApplication(
  userId: string,
  input: { postingId: string; resumeId: string | null; channel: Channel; appliedOn: string }
): Promise<void> {
  const posting = await getPosting(userId, input.postingId);
  if (!posting) throw new Error("找不到这个岗位");
  if (!posting.picked_at) await pickPosting(userId, posting.id);
  const { error } = await supabaseAdmin.from("job_applications").insert({
    user_id: userId,
    posting_id: input.postingId,
    resume_id: input.resumeId,
    channel: input.channel,
    status: "applied",
    events: [],
    applied_on: input.appliedOn,
  });
  if (error) {
    if (error.code === "23505") throw new Error("这个岗位已经记过投递了。");
    throw new Error(isMissingJobTables(error.message) ? MISSING_MIGRATION : error.message);
  }
}

/** 改状态：每次变化记下日期（"没回音"由代码按最后一次变化的日期算）。 */
export async function setApplicationStatus(
  userId: string,
  input: { id: string; status: ApplicationStatus; offeredSalary: number | null; today: string }
): Promise<void> {
  const { data, error } = await supabaseAdmin
    .from("job_applications")
    .select("status, events, offered_salary")
    .eq("id", input.id)
    .eq("user_id", userId)
    .maybeSingle();
  if (error) throw new Error(error.message);
  if (!data) throw new Error("找不到这条投递");
  const events = toEvents(data.events);
  const next = data.status === input.status ? events : [...events, { status: input.status, on: input.today }];
  const { error: updateError } = await supabaseAdmin
    .from("job_applications")
    .update({
      status: input.status,
      events: next,
      offered_salary: input.offeredSalary ?? (data.offered_salary as number | null) ?? null,
      updated_at: new Date().toISOString(),
    })
    .eq("id", input.id)
    .eq("user_id", userId);
  if (updateError) throw new Error(updateError.message);
}

/** "市场怎么看我"要用的：每条投递连同岗位的公司、职位、类别。 */
export async function listMarketApps(userId: string): Promise<MarketApp[]> {
  const { data, error } = await supabaseAdmin
    .from("job_applications")
    .select("posting_id, status, events, applied_on, offered_salary, job_postings(company_name, title, category)")
    .eq("user_id", userId);
  if (error) {
    if (isMissingJobTables(error.message)) return [];
    throw new Error(error.message);
  }
  return (data ?? []).map((r) => {
    const joined = r.job_postings as unknown;
    const p = (Array.isArray(joined) ? joined[0] : joined) as { company_name?: string; title?: string; category?: PostingCategory } | null;
    return {
      posting_id: r.posting_id as string,
      company_name: p?.company_name ?? "",
      title: p?.title ?? "",
      category: p?.category ?? "none",
      status: r.status as ApplicationStatus,
      events: toEvents(r.events),
      applied_on: r.applied_on as string,
      offered_salary: (r.offered_salary as number | null) ?? null,
    };
  });
}

/** 从"市场怎么看我"押一条进判断账本（source_type=self），到期他对着次数对账。 */
export async function createMarketBet(userId: string, input: { text: string; dueOn: string; confidence: number }): Promise<void> {
  const text = input.text.trim();
  if (text.length < 6) throw new Error("写清楚押的是什么，比如「数据分析类再投 5 家，至少 1 家约面试」。");
  if (!/^\d{4}-\d{2}-\d{2}$/.test(input.dueOn)) throw new Error("到期日格式不对");
  const confidence = Math.round(input.confidence);
  if (!Number.isFinite(confidence) || confidence < 0 || confidence > 100) throw new Error("把握度要在 0–100 之间");
  const { error } = await supabaseAdmin.from("predictions").insert({
    user_id: userId,
    source_type: "self",
    idea_id: null,
    period_id: null,
    hypothesis_id: null,
    text: `【求职】${text}`,
    due_at: `${input.dueOn}T23:59:59+09:00`,
    confidence,
  });
  if (error) throw new Error(error.message);
}

export function JobSearchLoader({ auto }: { auto: boolean }) {
  const router = useRouter();
  const [state, setState] = useState<"idle" | "running" | "failed" | "done">(auto ? "running" : "idle");

    setState("running");
    const r = await runJobSearchAction();
    if (!r.ok) {
      setMessage(r.message);
      setState("failed");
      return;
    }
    setMessage(
      `${r.kept > 0 ? `新留下 ${r.kept} 条。` : "今天没搜到新的。"}${r.skipped.length > 0 ? ` ${r.skipped.join("；")}。` : ""}`
    );
    setState("done");
    router.refresh();
  };

  useEffect(() => {
    if (!auto || started.current) return;
    started.current = true;
    void run();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  if (state === "running") {
    return <p className="text-sm leading-7 text-muted-foreground">在按你的条件搜四类岗位，一页页读招聘页……大概半分钟到一分钟。</p>;

  return (

      {message && <p>{state === "failed" ? `这次没搜成：${message}` : message}</p>}
      <Button variant="outline" size="sm" className="mt-2" onClick={() => void run()}>

import { notFound } from "next/navigation";
import { createClient } from "@/lib/supabase/server";
import { PageContainer } from "@/components/ui/page-container";
import { getPosting } from "@/lib/domains/jobs/postings";
import { getJobProfile } from "@/lib/domains/jobs/profile";
import { getResume } from "@/lib/domains/jobs/resumes";
import { EMPTY_PERSONAL, RESUME_SECTION_KEYS, RESUME_SECTIONS, sectionText, tokyoDate } from "@/lib/domains/jobs/shared";
import { PrintButton } from "./print-button";

export const dynamic = "force-dynamic";

/**
 * A4 打印页：浏览器"打印 → 存成 PDF"。没有导航壳（不在 app/(app) 下）。
 * 姓名、联系方式由代码从条件页的个人信息里填进来 —— 这些从来没发给过 AI。
 */
export default async function ResumePrintPage({ params }: { params: { rid: string } }) {
  const supabase = createClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();
  if (!user) notFound();
  if (!/^[0-9a-f-]{36}$/i.test(params.rid)) notFound();
  const resume = await getResume(user.id, params.rid);
  if (!resume) notFound();
  const [posting, profile] = await Promise.all([getPosting(user.id, resume.posting_id), getJobProfile(user.id)]);
  const personal = profile && profile !== "missing-schema" ? profile.personal : EMPTY_PERSONAL;
  const [y, m, d] = tokyoDate().split("-").map(Number);

  // narrow 档是 672px，正好等于 A4 减去两边 16mm 页边距的可打印宽度
  return (
    <PageContainer width="narrow" className="bg-background text-sm leading-7 text-foreground print:p-0">
      <style>{"@page { size: A4; margin: 16mm; }"}</style>
      <div className="mb-6 flex items-center justify-between print:hidden">
        <p className="text-xs text-muted-foreground">
          {posting ? `${posting.company_name} · ${posting.title}` : ""} · 第 {resume.version} 版
        </p>
        <PrintButton />
      </div>
      <h1 className="text-center text-xl font-semibold">職務経歴書</h1>
      <p className="mt-4 text-right">
        {y}年{m}月{d}日現在
      </p>
      <p className="text-right">
        氏名　{personal.name || "（条件页里还没填）"}
        {personal.kana && `（${personal.kana}）`}
      </p>
      {(personal.email || personal.phone) && (
        <p className="text-right text-xs">{[personal.email, personal.phone].filter(Boolean).join(" / ")}</p>
      )}
      {RESUME_SECTION_KEYS.filter((k) => k !== "motivation").map((key) => (
        <section key={key} className="mt-6 break-inside-avoid">
          <h2 className="border-b border-border pb-1 font-semibold">■ {RESUME_SECTIONS[key]}</h2>
          <p className="mt-2 whitespace-pre-line">{sectionText(key, resume.sections[key])}</p>
        </section>
      ))}
      <p className="mt-6 text-right">以上</p>
      <section className="mt-10 break-before-page">
        <h2 className="border-b border-border pb-1 font-semibold">■ {RESUME_SECTIONS.motivation}</h2>
        <p className="mt-2 whitespace-pre-line">{sectionText("motivation", resume.sections.motivation)}</p>
      </section>
    </PageContainer>
  );
}
